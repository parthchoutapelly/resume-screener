## 1. Guardrails

- Never render candidate-derived text as HTML: no `dangerouslySetInnerHTML`, no markdown rendering (R-SEC-08).
- The UI never treats **Recommended** as a decision, and never sends anything on its own (R-BUS-06).
- The shortlist confirmation always shows the destination email, or says none will be sent (Design §5.3).
- Hiding the Admin link is cosmetic; the server enforces access (R-AUTH-05). Still guard the route.
- File bytes go straight to S3 via presigned POST; they never pass through the API client.
- There are no secrets in the bundle. Pool ID, client ID, and API URL are public identifiers that come from stack outputs.

`vite.config.js`: in development the SPA calls `/api`, and Vite proxies it to API Gateway, so the API's CORS only has to allow the CloudFront origin:

import react from '@vitejs/plugin-react';
export default defineConfig(({ mode }) => {

  return {
    plugins: [react()],
    server: { port: 5173, proxy: { '/api': { target: env.VITE_API_PROXY_TARGET, changeOrigin: true,

Dev `.env.dev`: `VITE_API_BASE_URL=/api`, `VITE_API_PROXY_TARGET=<ApiBaseUrl>`. Prod `.env.prod`: `VITE_API_BASE_URL=<ApiBaseUrl>`. Both are generated by `scripts/gen-frontend-env.sh <env>` from stack outputs (T-090) and are never committed. Uploads in dev go straight to S3, which the upload bucket's CORS allows for `localhost:5173` in `dev` only.

`src/config.js` reads `import.meta.env` and **throws at startup** if a value is missing, so the app never runs half-configured.

## 3. Auth (T-081)

```js
// src/auth/cognito.js
import { Amplify } from 'aws-amplify';
import { signIn, confirmSignIn, signOut, fetchAuthSession, getCurrentUser } from 'aws-amplify/auth';
import { config } from '../config';

Amplify.configure({ Auth: { Cognito: { userPoolId: config.userPoolId, userPoolClientId: config.userPoolClientId } } });

export async function login(email, password) {
  const { isSignedIn, nextStep } = await signIn({ username: email, password });
  if (isSignedIn) return { done: true };
  if (nextStep.signInStep === 'CONFIRM_SIGN_IN_WITH_NEW_PASSWORD_REQUIRED') return { done: false, needsNewPassword: true };
  throw new Error(`Unsupported sign-in step: ${nextStep.signInStep}`);
}
export const completeNewPassword = (newPassword) => confirmSignIn({ challengeResponse: newPassword });
export const logout = () => signOut();

export async function getIdToken({ forceRefresh = false } = {}) {
  const s = await fetchAuthSession({ forceRefresh });
  return s.tokens?.idToken?.toString() ?? null;
}
export async function getGroups() {
  const s = await fetchAuthSession();
  return s.tokens?.idToken?.payload['cognito:groups'] ?? [];
}
```
`AuthContext` exposes `{ status: 'loading'|'signedOut'|'signedIn', user, groups, isAdmin, hasAccess, login, completeNewPassword, logout }`. `hasAccess = groups includes Recruiter or Admin`. When it's false, render `NoAccessPage` (Design §5.5) instead of the app. `ProtectedRoute` redirects to `/login?next=<path>`; `AdminRoute` renders a "You don't have access" page for non-admins.

## 4. API layer (T-082)

### 4.1 JSON client
```js
// src/api/client.js
import { config } from '../config';
import { getIdToken } from '../auth/cognito';

export class ApiError extends Error {
  constructor(status, code, message, details) { super(message); Object.assign(this, { status, code, details }); }
}

async function request(path, { method = 'GET', body, retried = false } = {}) {
  const token = await getIdToken({ forceRefresh: retried });
  const res = await fetch(`${config.apiBaseUrl}${path}`, {
    method,
    headers: { 'Content-Type': 'application/json', Authorization: token },   // raw ID token (03 §5)
    body: body === undefined ? undefined : JSON.stringify(body),
  });
  if (res.status === 401 && !retried) return request(path, { method, body, retried: true });  // one silent refresh
  const data = res.status === 204 ? null : await res.json().catch(() => null);
  if (!res.ok) {
    const e = data?.error ?? {};
    throw new ApiError(res.status, e.code ?? 'HTTP_' + res.status, e.message ?? 'Request failed', e.details);
  }
  return data;
}

export const api = {
  createJob:      (payload)          => request('/jobs', { method: 'POST', body: payload }),
  listJobs:       (nextToken)        => request('/jobs' + (nextToken ? `?next_token=${encodeURIComponent(nextToken)}` : '')),
  getJob:         (jobId)            => request(`/jobs/${jobId}`),
  updateJob:      (jobId, patch)     => request(`/jobs/${jobId}`, { method: 'PATCH', body: patch }),
  addResumes:     (jobId, filenames) => request(`/jobs/${jobId}/resumes`, { method: 'POST', body: { filenames } }),
  listCandidates: (jobId)            => request(`/jobs/${jobId}/candidates`),
  decide:         (jobId, cid, decision) => request(`/jobs/${jobId}/candidates/${cid}/decision`, { method: 'POST', body: { decision } }),
  resumeUrl:      (jobId, cid)       => request(`/jobs/${jobId}/candidates/${cid}/resume-url`),
  exportCsv:      (jobId)            => request(`/jobs/${jobId}/export`),
  listFailures:   ({ jobId, nextToken } = {}) =>
    request('/failed-jobs' + '?' + new URLSearchParams({ ...(jobId && { job_id: jobId }), ...(nextToken && { next_token: nextToken }) })),
};
```
A second 401 after the refresh → `logout()` and redirect to `/login?next=…` with "Your session expired."

### 4.2 Presigned-POST uploads (T-083)
`fetch` has no upload progress events, so this uses XHR:
```js
// src/api/upload.js
export class UploadError extends Error { constructor(status, kind) { super(kind); this.status = status; this.kind = kind; } }

export function uploadFile({ url, fields }, file, onProgress) {
  return new Promise((resolve, reject) => {
    const form = new FormData();
    Object.entries(fields).forEach(([k, v]) => form.append(k, v));
    form.append('file', file);                                   // MUST be the last field for S3 POST
    const xhr = new XMLHttpRequest();
    xhr.open('POST', url);
    xhr.upload.onprogress = (e) => e.lengthComputable && onProgress(e.loaded / e.total);
    xhr.onload = () => {
      if (xhr.status >= 200 && xhr.status < 300) return resolve();          // S3 returns 204
      const body = xhr.responseText || '';
      const kind = /Policy expired|Request has expired/i.test(body) ? 'expired'
                 : /EntityTooLarge/i.test(body) ? 'too_large' : 'rejected';
      reject(new UploadError(xhr.status, kind));
    };
    xhr.onerror = () => reject(new UploadError(0, 'network'));
    xhr.send(form);
  });
}
```
`useUploadQueue(items)` runs a pool of **4** concurrent uploads. It retries a `network` failure once automatically. `expired` never retries: the row tells the user to re-upload from the job page, which calls `api.addResumes` for new credentials. The hook exposes per-file `{status: queued|uploading|uploaded|failed, progress, error}` and totals, and sets a `beforeunload` guard while anything is in flight.

Client-side pre-validation (R-VAL-04) mirrors the server: extensions `pdf, docx, png, jpg, jpeg, tiff` (case-insensitive), ≤10 MiB, ≤50 files, JD file same extensions. Invalid files show their reason inline and are excluded (Design §4.3).

## 6. Polling (T-087, D-44)

`usePolling(fetchFn, { intervalMs: 5000, shouldPoll, maxDurationMs: 30 * 60 * 1000 })`:
- Runs while `shouldPoll(data)` is true: any candidate in `IN_PROGRESS`, or the job's `blocking_reason === 'awaiting_jd'`.
- Pauses on `document.visibilityState === 'hidden'` and resumes (with an immediate fetch) on `visible`.
- After 3 consecutive failures, backs off to 15 s and shows "Having trouble refreshing". It keeps the last good data and never blanks the table.
- Stops after `maxDurationMs` with an "Auto-refresh paused — Refresh" affordance.
- Merges results by `candidate_id` and keeps expanded rows and focus. **Re-sorting is deferred while the pointer is over the table** (Design §5.2).
- Announces batch changes through one polite live region ("3 candidates scored").
- There is exactly **one** timer per page, cleared on unmount.

## 9. Release (T-090–T-092)

```bash
# scripts/gen-frontend-env.sh <env>   → frontend/.env.<env> from `aws cloudformation describe-stacks` outputs
# scripts/deploy-frontend.sh <env>
set -euo pipefail
ENV="$1"; STACK="resume-screener-$ENV"
out() { aws cloudformation describe-stacks --stack-name "$STACK" --query "Stacks[0].Outputs[?OutputKey=='$1'].OutputValue" --output text; }
WEB=$(out WebBucketName); DIST=$(out DistributionId)
(cd frontend && npm ci && npm run build -- --mode "$ENV")
aws s3 sync frontend/dist/ "s3://$WEB/" --delete --exclude index.html --cache-control "public,max-age=31536000,immutable"
aws s3 cp frontend/dist/index.html "s3://$WEB/index.html" --cache-control "no-cache"
aws cloudfront create-invalidation --distribution-id "$DIST" --paths "/index.html" >/dev/null
echo "Deployed: https://$(out DistributionDomain)"
```
There's no S3 website hosting and no Hosted UI (the app uses direct `signIn`), so no Cognito callback URLs are needed. CORS for the API and the upload bucket is derived from the distribution domain in the template (phases 1 and 3), so there's nothing to edit by hand after the first deploy (T-092 is verification only).
